# basic: buses, 2D mono/stereo, volume, mute, bus volume, stop with fade, limiter overload
rate 48000
voices 64
bus master
bus sfx master
bus music master
bus ui master
event dc sfx dc_half.wav loop=1
event dc_half_volume sfx dc_half.wav loop=1 volume=0.5
event dc_music music dc_half.wav loop=1
event sine sfx sine440.wav loop=1
event stereo sfx stereo.wav loop=1
event beep ui beep.wav
event step sfx step_a.wav,step_b.wav,step_c.wav vol_rand=0.2 pitch_rand=0.1
event wind sfx wind_loop.wav loop=1 lowpass=2000
play dc
render 1000
play sine
play stereo
render 3000
setvol 1 0.5
setpitch 2 1.5
render 2000
mute sfx 1
render 1000
mute sfx 0
busvol music 0.25
play dc_music
play beep ui
render 2000
play step
play step
play step
render 4000
stop 0 0.01
play wind
render 2000
update
stopall 0.02
render 3000
