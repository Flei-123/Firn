#name box shorthands, the one to four value rule
#html
<!DOCTYPE html><html><body><div id="a"></div><div id="b"></div><div id="c"></div><div id="d"></div></body></html>
#author
#a { margin: 5px }
#b { margin: 5px 10px }
#c { margin: 5px 10px 15px }
#d { margin: 5px 10px 15px 20px; padding: 1px 2px; border: 3px dashed red }
#d { margin-left: 7px }
#match
// 0 html  1 head  2 body  3 div#a  4 div#b  5 div#c  6 div#d
div | 3 4 5 6
#style
3 | margin=5.000,5.000,5.000,5.000
4 | margin=5.000,10.000,5.000,10.000
5 | margin=5.000,10.000,15.000,10.000
// The shorthand becomes four longhands, so the later `margin-left` beats
// the left value of the shorthand -- and only that one.
6 | margin=5.000,10.000,15.000,7.000 padding=1.000,2.000,1.000,2.000 border-width=3.000,3.000,3.000,3.000 border-style=dashed,dashed,dashed,dashed border-color=ffff0000,ffff0000,ffff0000,ffff0000
#end
